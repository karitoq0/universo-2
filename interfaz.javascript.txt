/* =========================================================
   ESTRELLAS
========================================================= */

function createSpace(){

  const stars = $("stars");

  stars.innerHTML = "";

  for(let i=0;i<220;i++){

    const star = document.createElement("span");

    star.className = "star";

    star.style.left = Math.random()*100 + "%";
    star.style.top = Math.random()*100 + "%";

    const size = Math.random()*2+1;

    star.style.width = size+"px";
    star.style.height = size+"px";

    star.style.animationDelay =
      Math.random()*3+"s";

    stars.appendChild(star);
  }
}


/* =========================================================
   SELECTOR
========================================================= */

function createSelector(){

  planetSelector.innerHTML = "";

  planets.forEach((p,index)=>{

    const button =
      document.createElement("button");

    button.className =
      "planet-select-button";

    button.type = "button";

    button.textContent =
      p.name;

    button.dataset.index =
      index;

    button.addEventListener(
      "click",
      ()=>{
        selectPlanet(index);
      }
    );

    planetSelector.appendChild(button);

  });

}


/* =========================================================
   SELECCIONAR PLANETA
========================================================= */

function selectPlanet(index,notify=true){

  const p = planets[index];

  if(!p) return;

  currentPlanetIndex = index;

  visitedPlanets.add(index);

  $("planetName").textContent =
    p.name;

  $("planetLabel").textContent =
    p.name.toUpperCase();

  $("planetType").textContent =
    p.type;

  $("planetDescription").textContent =
    p.description;

  $("habitabilityBadge").textContent =
    p.badge;

  $("habitabilityBadge").className =
    "status-badge "+p.habitability;

  $("calculatorPlanet").textContent =
    p.name.toUpperCase();


  /* PLANETA VISUAL */

  planet.style.background =
    p.gradient;

  planet.style.boxShadow =
    `inset -30px -20px 50px rgba(0,0,0,.55),
     0 0 80px ${p.glow}`;

  planet.className =
    "planet";

  if(p.ring)
    planet.classList.add("has-ring");

  if(p.moon)
    planet.classList.add("has-moon");


  /* STATS */

  $("statsGrid").innerHTML =
    p.stats.map(s=>`

      <div class="stat">

        <span>${s[0]}</span>

        <strong>${s[1]}</strong>

      </div>

    `).join("");


  $("funFact").textContent =
    p.fact;


  $("eventsList").innerHTML =
    p.events.map(event=>`

      <div class="event-item">
        ⚠️ ${event}
      </div>

    `).join("");


  $("missionScore").textContent =
    p.score+"%";

  $("missionProgress").style.width =
    p.score+"%";

  $("missionSummary").textContent =
    p.summary;


  $("equipmentList").innerHTML =
    p.equipment.map(item=>`

      <div class="equipment-item">
        🧰 ${item}
      </div>

    `).join("");


  $("lifeText").textContent =
    p.life;


  updateExploration();

  updateMissionLog();

  updateAchievements();

  calculateTransformation();

  if(notify){

    showToast(
      "🛰️ Destino seleccionado: "+p.name
    );

    playTone();

  }

}


/* =========================================================
   EXPLORACIÓN
========================================================= */

function updateExploration(){

  const percentage =
    Math.round(
      visitedPlanets.size /
      planets.length * 100
    );

  $("explorationPercent").textContent =
    percentage+"%";

  $("explorationBar").style.width =
    percentage+"%";

  document
    .querySelectorAll(".planet-select-button")
    .forEach(button=>{

      button.classList.toggle(
        "active",
        Number(button.dataset.index)
        === currentPlanetIndex
      );

    });

}


/* =========================================================
   REGISTRO
========================================================= */

function updateMissionLog(){

  const count =
    visitedPlanets.size;

  $("visitedCount").textContent =
    `${count} / ${planets.length}`;

  $("currentMissionPlanet").textContent =
    planets[currentPlanetIndex].name;


  let rank =
    "Cadete espacial";

  if(count>=3)
    rank="Navegante orbital";

  if(count>=5)
    rank="Explorador planetario";

  if(count===8)
    rank="Comandante del sistema solar";

  $("explorerRank").textContent =
    rank;

}

/* =========================================================
   ROTACIÓN
========================================================= */

$("toggleRotation")
.addEventListener(
  "click",
  ()=>{

    rotationPaused =
      !rotationPaused;

    planet.classList.toggle(
      "paused",
      rotationPaused
    );

    $("toggleRotation").textContent =
      rotationPaused
      ? "▶ REANUDAR"
      : "⏸ PAUSAR ROTACIÓN";

  }
);


$("rotateLeft")
.addEventListener(
  "click",
  ()=>{

    rotationDegrees -= 25;

    planet.style.transform =
      `rotate(${rotationDegrees}deg)`;

  }
);


$("rotateRight")
.addEventListener(
  "click",
  ()=>{

    rotationDegrees += 25;

    planet.style.transform =
      `rotate(${rotationDegrees}deg)`;

  }
);


/* =========================================================
   ARRASTRAR
========================================================= */

planet.addEventListener(
  "pointerdown",
  e=>{

    dragging=true;

    dragStartX=e.clientX;

    dragStartRotation=
      rotationDegrees;

    planet.setPointerCapture(
      e.pointerId
    );

  }
);

planet.addEventListener(
  "pointermove",
  e=>{

    if(!dragging) return;

    const diff =
      e.clientX-dragStartX;

    rotationDegrees =
      dragStartRotation+diff;

    planet.style.transform =
      `rotate(${rotationDegrees}deg)`;

  }
);

planet.addEventListener(
  "pointerup",
  ()=>{
    dragging=false;
  }
);


/* =========================================================
   CALCULADORA
========================================================= */

function calculateTransformation(){

  const weight =
    Number($("earthWeight").value);

  const age =
    Number($("earthAge").value);

  const p =
    planets[currentPlanetIndex];

  if(!weight || !age) return;

  const newWeight =
    weight*p.gravity;

  const newAge =
    age/p.orbitalPeriod;

  $("weightResult").textContent =
    newWeight.toFixed(1)+" kg";

  $("ageResult").textContent =
    newAge.toFixed(1)+" años";

}

$("calculateButton")
.addEventListener(
  "click",
  ()=>{

    calculateTransformation();

    showToast(
      "🧪 Simulación completada"
    );

    playTone();

  }
);

function generateEvent(){

  const p =
    planets[currentPlanetIndex];

  const event =
    p.events[

      )
    ];

  $("eventTitle").textContent =
    event;

  showToast(
    "⚡ Evento detectado"
  );

  playTone();

}


/* =========================================================
   MODALES
========================================================= */

function openModal(title,content){

  modalTitle.textContent =
    title;

  modalContent.innerHTML =
    content;

  modal.classList.add("active");

  document.body.style.overflow =
    "hidden";

}

function closeModal(){

  modal.classList.remove("active");

  document.body.style.overflow =
    "";

}

$("closeModal")
.addEventListener(
  "click",
  closeModal
);

modal.addEventListener(
  "click",
  e=>{

    if(e.target===modal)
      closeModal();

  }
);

document.addEventListener(
  "keydown",
  e=>{

    if(e.key==="Escape")
      closeModal();

  }
);

/* =========================================================
   SONIDO
========================================================= */

function playTone(){

  if(!soundEnabled)
    return;

  try{

    if(!audioContext){

      audioContext =
        new(
          window.AudioContext ||
          window.webkitAudioContext
        )();

    }

    const oscillator =
      audioContext.createOscillator();

    const gain =
      audioContext.createGain();

    oscillator.type="sine";

    oscillator.frequency.value=520;

    gain.gain.setValueAtTime(
      .05,
      audioContext.currentTime
    );

    gain.gain.exponentialRampToValueAtTime(
      .001,
      audioContext.currentTime+.16
    );

    oscillator.connect(gain);
    gain.connect(audioContext.destination);

    oscillator.start();

    oscillator.stop(
      audioContext.currentTime+.16
    );

  }catch(error){

    console.log(
      "Audio no disponible"
    );

  }

}


$("soundButton")
.addEventListener(
  "click",
  ()=>{

    soundEnabled =
      !soundEnabled;

    $("soundButton").textContent =
      soundEnabled ? "🔊":"🔇";

    if(soundEnabled)
      playTone();

    showToast(
      soundEnabled
      ? "🔊 Sonido activado"
      : "🔇 Sonido desactivado"
    );

  }
);

    function conectarBotonesSimulacion() {

      const mapaButton =
        document.querySelector("[data-open-holo-map]");

      if (mapaButton) {

        mapaButton.addEventListener("click", function () {
          abrirMapaHolografico();
        });

      }

      const nextButton =
        document.querySelector("[data-sim-next]");

      if (nextButton) {

        nextButton.addEventListener("click", function () {

          const actual = obtenerIndiceActual();

          const siguiente =
            (actual + 1) % datosPlanetas.length;

          abrirSimulacion(siguiente);

          try {
            if (typeof selectPlanet === "function") {
              selectPlanet(siguiente, true);
            }
          } catch (error) {}

        });

      }
    }

      };

      return escenarios[planeta.id] ||

    function calcularViabilidad(planeta) {

      if (planeta.id === "earth") return 95;
      if (planeta.id === "mars") return 42;
      if (planeta.id === "venus") return 8;
      if (planeta.id === "mercury") return 12;
      if (planeta.id === "saturn") return 5;
      if (planeta.id === "jupiter") return 4;
      if (planeta.id === "uranus") return 3;
      if (planeta.id === "neptune") return 2;

      return 10;
    }

    function obtenerIndiceActual() {

      try {

        if (
          typeof currentPlanetIndex !== "undefined" &&
          Number.isInteger(currentPlanetIndex)
        ) {
          return currentPlanetIndex;
        }

      } catch (error) {}

      return 0;
    }

       CONECTAR LOS BOTONES QUE YA EXISTEN
       ===================================================== */

    function conectarBotonesExistentes() {

      const elementos =
        document.querySelectorAll(
          "button, a, [role='button']"
        );

      elementos.forEach(elemento => {

        const texto =
          elemento.textContent
            .trim()
            .toLowerCase()
            .replace(/\s+/g, " ");

        /*
         * Busca el botón por su texto.
         * Así no importa qué ID tenía en el código anterior.
         */

        if (
          texto.includes("abrir mapa holográfico") ||
          texto.includes("mapa holográfico")
        ) {

          elemento.addEventListener("click", function (event) {

            event.preventDefault();

            abrirMapaHolografico();

          });

        }

        if (
          texto.includes("simulación humana") ||
          texto.includes("simulacion humana")
        ) {

          elemento.addEventListener("click", function (event) {

            event.preventDefault();

            abrirSimulacion(
              obtenerIndiceActual()
            );

          });

        }

      });
    }

    /*
     * También usamos delegación para botones que el código
     * anterior pueda crear dinámicamente.
     */

    document.addEventListener("click", function (event) {

      const mapa =
        event.target.closest("[data-open-holographic-map]");

      if (mapa) {
        event.preventDefault();
        abrirMapaHolografico();
      }

      const simulacion =
        event.target.closest("[data-open-human-simulation]");

      if (simulacion) {
        event.preventDefault();
        abrirSimulacion(obtenerIndiceActual());
      }

    });

    conectarBotonesExistentes();

    /* -----------------------------------------------------
       Exponer funciones para que el código anterior pueda
       llamarlas si lo necesita.
       ----------------------------------------------------- */

    window.abrirMapaHolografico = abrirMapaHolografico;
    window.abrirSimulacionHumana = function () {
      abrirSimulacion(obtenerIndiceActual());
    };
